TABLE 1: profiles (User Accounts)

id: (UUID) Primary Key

full_name: (Text) Name of Ali or other users

role: (Text) 'worker', 'verifier', 'advocate', or 'analyst' (advocate and analyst share app permissions)

city_zone: (Text) e.g., 'Faisal Town', 'Gulberg'

created_at: (Timestamp)

TABLE 2: earnings (Ali’s Shift Logs)

id: (UUID) Primary Key

worker_id: (UUID) Links to profiles(id)

platform: (Text) e.g., 'Uber', 'FoodPanda', 'Bykea'

date: (Date)

hours_worked: (Number)

gross_earned: (Number) Total before cuts

deductions: (Number) Commission taken by app

net_received: (Number) Final amount paid

screenshot_url: (Text) URL of uploaded proof

status: (Text) 'pending', 'verified', 'flagged', or 'unverifiable'

anomaly_explanation: (Text) Reason from AI/FastAPI

created_at: (Timestamp)

TABLE 3: grievances (Complaint Board)

id: (UUID) Primary Key

worker_id: (UUID) Links to profiles(id)

platform: (Text)

category: (Text) e.g., 'Payment', 'Deactivation'

description: (Text) The actual complaint text

status: (Text) 'open', 'escalated', or 'resolved'

tags: (Array/Text) Used for clustering/grouping

created_at: (Timestamp)

TABLE 4: market_benchmarks (Seed Data)

id: (UUID) Primary Key

platform: (Text)

city_zone: (Text)

avg_hourly_pay: (Number) e.g., 400

avg_deduction_rate: (Number) e.g., 0.20 (20%)

created_at: (Timestamp)

TABLE 5: community_posts (Worker bulletin; anonymous in feed)

id: (UUID) Primary Key

author_id: (UUID) Links to profiles(id); omitted in GET /api/community/feed

title: (Text)

body: (Text)

platform: (Text, optional)

category: (Text) default 'general'

tags: (Text array)

status: (Enum) 'pending' | 'visible' | 'hidden' | 'removed' — advocates set visible/hidden/removed only via API

moderator_note: (Text, optional)

created_at, updated_at: (Timestamp)

TABLE 6: community_post_upvotes

post_id: (UUID) FK → community_posts(id), part of PK

user_id: (UUID) FK → profiles(id), part of PK

created_at: (Timestamp)

TABLE 7: community_post_comments

id: (UUID) Primary Key

post_id: (UUID) FK → community_posts(id)

author_id: (UUID) FK → profiles(id) — not returned in public comment API

body: (Text)

created_at: (Timestamp)

TABLE 8: anomaly_runs (Anomaly service audit)

id: (UUID) Primary Key

created_at: (Timestamp)

worker_id: (UUID) FK → profiles(id)

shift_log_id: (UUID, optional) FK → earnings(id)

triggered_by_user_id: (UUID, optional) FK → profiles(id)

triggered_by_role: (Text)

platform: (Text, optional)

current_shift_date: (Date)

ready: (Boolean) — false when insufficient verified history

insufficient_reason: (Text, optional)

is_anomaly: (Boolean, optional) — null when not ready

explanation: (Text, optional)

method: (Text, optional)

data_source: (Text, optional)

net_z_score, percent_drop: (Numbers, optional)

is_net_anomaly: (Boolean, optional)

deduction_share_z_score, current_deduction_ratio, mean_deduction_ratio: (Numbers, optional)

is_deduction_anomaly: (Boolean, optional)

history_count: (Integer)

thresholds: (JSONB, optional)

detail: (JSONB, optional)
